Function main
  Assign
    Var(total)
    Const(0,DataType.INT)
  Assign
    Var(count)
    Const(0,DataType.INT)
  While
    RelOp(<)
      Var(count)
      Const(4,DataType.INT)
    Block
      Assign
        Var(count)
        BinOp(+)
          Var(count)
          Const(1,DataType.INT)
      Assign
        Var(total)
        BinOp(+)
          Var(total)
          BinOp(*)
            Var(count)
            Const(10,DataType.INT)
  Assign
    Var(avg)
    BinOp(/)
      Cast(DataType.DOUBLE)
        Var(total)
      Var(count)
  IfElse
    RelOp(>=)
      Var(avg)
      Const(30.0,DataType.DOUBLE)
    Assign
      Var(level)
      Const(high,DataType.STRING)
    IfElse
      RelOp(>=)
        Var(avg)
        Const(20.0,DataType.DOUBLE)
      Assign
        Var(level)
        Const(medium,DataType.STRING)
      Assign
        Var(level)
        Const(low,DataType.STRING)
  Print
    Var(avg)
  Print
    Var(level)
