.data
g_total: .word 0
.text
.globl main
f_bump:
    subu $sp, $sp, 4
    sw   $ra, 0($sp)
    subu $sp, $sp, 4
    sw   $fp, 0($sp)
    addiu $fp, $sp, -0
    move $sp, $fp
    lw $t0, g_total
    li $t1, 1
    add $t2, $t0, $t1
    sw $t2, g_total
    addiu $sp, $fp, 0
    lw    $fp, 0($sp)
    addiu $sp, $sp, 4
    lw    $ra, 0($sp)
    addiu $sp, $sp, 4
    jr    $ra
f_compute:
    subu $sp, $sp, 4
    sw   $ra, 0($sp)
    subu $sp, $sp, 4
    sw   $fp, 0($sp)
    addiu $fp, $sp, -20
    move $sp, $fp
    sw $a0, 16($fp)
    lw $t0, 16($fp)
    li $t1, 100
    mul $t2, $t0, $t1
    sw $t2, 0($fp)
    lw $t0, 16($fp)
    li $t1, 2
    mul $t2, $t0, $t1
    sw $t2, 4($fp)
    lw $t0, 16($fp)
    li $t1, 1
    add $t2, $t0, $t1
    sw $t2, 4($fp)
    lw $t0, g_total
    li $t1, 2
    mul $t2, $t0, $t1
    sw $t2, 8($fp)
    jal f_bump
    lw $t0, g_total
    li $t1, 2
    mul $t2, $t0, $t1
    sw $t2, 12($fp)
    lw $t0, 4($fp)
    lw $t1, 8($fp)
    add $t2, $t0, $t1
    lw $t0, 12($fp)
    add $t1, $t2, $t0
    move $v0, $t1
    j ret_compute
    ret_compute:
    addiu $sp, $fp, 20
    lw    $fp, 0($sp)
    addiu $sp, $sp, 4
    lw    $ra, 0($sp)
    addiu $sp, $sp, 4
    jr    $ra
main:
    subu $sp, $sp, 4
    sw   $ra, 0($sp)
    subu $sp, $sp, 4
    sw   $fp, 0($sp)
    addiu $fp, $sp, -8
    move $sp, $fp
    li $t0, 10
    sw $t0, g_total
    li $t0, 99
    sw $t0, 4($fp)
    li $t0, 4
    move $a0, $t0
    jal f_compute
    move $t0, $v0
    sw $t0, 0($fp)
    lw $t0, 0($fp)
    li $t1, 1
    add $t2, $t0, $t1
    move $a0, $t2
    li $v0, 1
    syscall
    li $t0, 0
    move $a0, $t0
    jal f_compute
    move $t0, $v0
    sw $t0, 0($fp)
    lw $t0, g_total
    move $a0, $t0
    li $v0, 1
    syscall
    addiu $sp, $fp, 8

    addiu $sp, $sp, 4
    lw    $ra, 0($sp)

    jr    $ra
